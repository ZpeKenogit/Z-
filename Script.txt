-- =========================================================================
-- HELPER: CRIAR INSTÂNCIA COM PROPRIEDADES
-- =========================================================================
local function Create(className, props, children)
    local obj = Instance.new(className)
    for k, v in pairs(props or {}) do
        obj[k] = v
    end
    for _, child in ipairs(children or {}) do
        child.Parent = obj
    end
    return obj
end

local function AddCorner(parent, radius)
    return Create("UICorner", { CornerRadius = UDim.new(0, radius or 8), Parent = parent })
end

local function AddStroke(parent, color, thickness, transparency)
    return Create("UIStroke", {
        Color = color or Colors.Stroke,
        Thickness = thickness or 1,
        Transparency = transparency or 0.3,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = parent
    })
end

-- FUNÇÃO DE TROCA DE ABA
-- =========================================================================
local activeTab = "AIMBOT"
local function SwitchTab(tabName)
    activeTab = tabName
    for name, data in pairs(tabBtns) do
        if name == tabName then
            data.button.BackgroundColor3 = Colors.BgCard
            data.stroke.Color = Colors.Accent
            data.stroke.Transparency = 0
            data.label.TextColor3 = Colors.Accent
        else
            data.button.BackgroundColor3 = Colors.BgPanel
            data.stroke.Color = Colors.Stroke
            data.stroke.Transparency = 0.5
            data.label.TextColor3 = Colors.TextSecondary
        end
    end
    for name, container in pairs(tabContents) do
        container.Visible = (name == tabName)
    end
end

for name, data in pairs(tabBtns) do
    data.button.MouseButton1Click:Connect(function()
        SwitchTab(name)
    end)
end

-- FÁBRICA DE TOGGLE ROW (nome + descrição + toggle switch)
-- =========================================================================
local function MakeToggleRow(parent, yOffset, title, desc, initialValue, callback)
    local row = Create("Frame", {
        Size = UDim2.new(1, -24, 0, 52),
        Position = UDim2.new(0, 12, 0, yOffset),
        BackgroundTransparency = 1,
        Parent = parent
    })

    Create("TextLabel", {
        Size = UDim2.new(1, -60, 0, 16),
        Position = UDim2.new(0, 0, 0, 6),
        BackgroundTransparency = 1,
        Text = title,
        TextColor3 = Colors.TextPrimary,
        TextSize = 13,
        Font = Enum.Font.GothamSemibold,
        TextXAlignment = Enum.TextXAlignment.Left,
        Parent = row
    })

    Create("TextLabel", {
        Size = UDim2.new(1, -60, 0, 14),
        Position = UDim2.new(0, 0, 0, 24),
        BackgroundTransparency = 1,
        Text = desc,
        TextColor3 = Colors.TextDim,
        TextSize = 10,
        Font = Enum.Font.Gotham,
        TextXAlignment = Enum.TextXAlignment.Left,
        Parent = row
    })

    -- Toggle switch
    local switchBg = Create("Frame", {
        Size = UDim2.new(0, 42, 0, 22),
        Position = UDim2.new(1, -42, 0.5, 0),
        AnchorPoint = Vector2.new(0, 0.5),
        BackgroundColor3 = initialValue and Colors.ToggleOn or Colors.ToggleOff,
        BorderSizePixel = 0,
        Parent = row
    })
    AddCorner(switchBg, 11)

    local knob = Create("Frame", {
        Size = UDim2.new(0, 18, 0, 18),
        Position = initialValue and UDim2.new(1, -20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9),
        BackgroundColor3 = Color3.fromRGB(255, 255, 255),
        BorderSizePixel = 0,
        Parent = switchBg
    })
    AddCorner(knob, 9)

    local switchBtn = Create("TextButton", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text = "",
        Parent = switchBg
    })

    local state = initialValue
    switchBtn.MouseButton1Click:Connect(function()
        state = not state
        if state then
            TweenService:Create(switchBg, TweenInfo.new(0.2), { BackgroundColor3 = Colors.ToggleOn }):Play()
            TweenService:Create(knob, TweenInfo.new(0.2), {
                Position = UDim2.new(1, -20, 0.5, -9)
            }):Play()
        else
            TweenService:Create(switchBg, TweenInfo.new(0.2), { BackgroundColor3 = Colors.ToggleOff }):Play()
            TweenService:Create(knob, TweenInfo.new(0.2), {
                Position = UDim2.new(0, 2, 0.5, -9)
            }):Play()
        end
        if callback then callback(state) end
    end)

    return row
end

local function MakeSliderRow(parent, yOffset, title, desc, minVal, maxVal, initialValue, valueName, callback)
    local row = Create("Frame", {
        Size = UDim2.new(1, -24, 0, 78),
        Position = UDim2.new(0, 12, 0, yOffset),
        BackgroundTransparency = 1,
        Parent = parent
    })

    Create("TextLabel", {
        Size = UDim2.new(1, -100, 0, 16),
        Position = UDim2.new(0, 0, 0, 4),
        BackgroundTransparency = 1,
        Text = title,
        TextColor3 = Colors.TextPrimary,
        TextSize = 13,
        Font = Enum.Font.GothamSemibold,
        TextXAlignment = Enum.TextXAlignment.Left,
        Parent = row
    })

    -- Caixa de valor
    local valueBox = Create("Frame", {
        Size = UDim2.new(0, 90, 0, 22),
        Position = UDim2.new(1, -90, 0, 2),
        BackgroundColor3 = Colors.BgCard,
        BorderSizePixel = 0,
        Parent = row
    })
    AddCorner(valueBox, 6)
    AddStroke(valueBox, Colors.Stroke, 1, 0.4)

    local valueLabel = Create("TextLabel", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text = tostring(initialValue) .. " " .. valueName,
        TextColor3 = Colors.Accent,
        TextSize = 11,
        Font = Enum.Font.GothamBold,
        Parent = valueBox
    })

    Create("TextLabel", {
        Size = UDim2.new(1, -60, 0, 14),
        Position = UDim2.new(0, 0, 0, 22),
        BackgroundTransparency = 1,
        Text = desc,
        TextColor3 = Colors.TextDim,
        TextSize = 10,
        Font = Enum.Font.Gotham,
        TextXAlignment = Enum.TextXAlignment.Left,
        Parent = row
    })

    -- Slider bar
    local sliderBg = Create("Frame", {
        Size = UDim2.new(1, 0, 0, 5),
        Position = UDim2.new(0, 0, 0, 56),
        BackgroundColor3 = Colors.ToggleOff,
        BorderSizePixel = 0,
        Parent = row
    })
    AddCorner(sliderBg, 3)

    local pct = (initialValue - minVal) / (maxVal - minVal)
    local sliderFill = Create("Frame", {
        Size = UDim2.new(pct, 0, 1, 0),
        BackgroundColor3 = Colors.Accent,
        BorderSizePixel = 0,
        Parent = sliderBg
    })
    AddCorner(sliderFill, 3)

    local knob = Create("Frame", {
        Size = UDim2.new(0, 14, 0, 14),
        Position = UDim2.new(pct, -7, 0.5, -7),
        BackgroundColor3 = Colors.TextPrimary,
        BorderSizePixel = 0,
        ZIndex = 2,
        Parent = sliderBg
    })
    AddCorner(knob, 7)

    -- Botão invisível para arrastar
    local trackBtn = Create("TextButton", {
        Size = UDim2.new(1, 0, 0, 30),
        Position = UDim2.new(0, 0, 0.5, -15),
        BackgroundTransparency = 1,
        Text = "",
        Parent = sliderBg
    })

    local dragging = false
    local value = initialValue

    local function updateFromInput(input)
        local relX = math.clamp((input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X, 0, 1)
        value = math.floor(minVal + relX * (maxVal - minVal) + 0.5)
        sliderFill.Size = UDim2.new(relX, 0, 1, 0)
        knob.Position = UDim2.new(relX, -7, 0.5, -7)
        valueLabel.Text = tostring(value) .. " " .. valueName
        if callback then callback(value) end
    end

    trackBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateFromInput(input)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromInput(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    return row
end
