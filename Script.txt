-- Tabela de IDs permitidos para o Blox Fruits (Sea 1, Sea 2 e Sea 3)
local ALLOWED_PLACES = {
    [2753915549] = true, -- First Sea
    [4442272183] = true, -- Second Sea
    [7449423635] = true  -- Third Sea
}

-- Verifica se o jogador está em algum dos mares do Blox Fruits
if not ALLOWED_PLACES[game.PlaceId] then
    pcall(function()
        game:GetService("StarterGui"):SetCore("SendNotification", {
            Title = "ZpeKeno Hub",
            Text = "Erro: Este script so funciona no Blox Fruits!",
            Duration = 6
        })
    end)
    return -- Para a execução se não estiver no Blox Fruits
end

-- Tenta carregar a biblioteca de interface (Orion Library)
local OrionLib
local success, err = pcall(function()
    OrionLib = loadstring(game:HttpGet('https://raw.githubusercontent.com/jensonhirst/Orion/main/source'))()
end)

if not success or not OrionLib then
    warn("Falha ao carregar Orion Library: " .. tostring(err))
    return
end

-- =========================================================================
-- LOGIC LOOPS (Ajustes de Movimentação do Personagem)
-- =========================================================================
RunService.RenderStepped:Connect(function()
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
        local Humanoid = LocalPlayer.Character.Humanoid
        
        if Settings.Other.MovementSpeed and Settings.Other.MovementSpeed > 16 then
            Humanoid.WalkSpeed = Settings.Other.MovementSpeed
        end
        
        if Settings.Other.JumpHeight and Settings.Other.JumpHeight > 50 then
            if Humanoid.UseJumpPower then
                Humanoid.JumpPower = Settings.Other.JumpHeight
            else
                Humanoid.JumpHeight = Settings.Other.JumpHeight
            end
        end
    end
end)

-- =========================================================================
-- FUNÇÃO TOGGLE & BOTÃO MÓVEL RESPONSIVO
-- =========================================================================
local function ToggleHub()
    local ui = CoreGui:FindFirstChild("Orion") or (LocalPlayer:FindFirstChild("PlayerGui") and LocalPlayer.PlayerGui:FindFirstChild("Orion"))
    if ui then
        ui.Enabled = not ui.Enabled
    end
end

-- Atalho no teclado (Tecla Z)
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not gameProcessed and input.KeyCode == Enum.KeyCode.Z then
        ToggleHub()
    end
end)

-- Criar Botão Flutuante 'Z' para Celular / Tablet
local function CreateMobileButton()
    local ScreenGui = Instance.new("ScreenGui")
    ScreenGui.Name = "ZpeKenoMobileUI"
    ScreenGui.ResetOnSpawn = false
    
    pcall(function() ScreenGui.Parent = CoreGui end)
    if not ScreenGui.Parent then
        ScreenGui.Parent = LocalPlayer:WaitForChild("PlayerGui")
    end

    local OpenBtn = Instance.new("TextButton")
    OpenBtn.Name = "ToggleZButton"
    OpenBtn.Parent = ScreenGui
    OpenBtn.Size = UDim2.new(0, 50, 0, 50)
    OpenBtn.Position = UDim2.new(0, 15, 0.35, 0)
    OpenBtn.BackgroundColor3 = Color3.fromRGB(15, 23, 42)
    OpenBtn.BorderColor3 = Color3.fromRGB(0, 170, 255)
    OpenBtn.BorderSizePixel = 2
    OpenBtn.Text = "Z"
    OpenBtn.TextColor3 = Color3.fromRGB(0, 170, 255)
    OpenBtn.TextSize = 22
    OpenBtn.Font = Enum.Font.SourceSansBold
    OpenBtn.Active = true
    OpenBtn.Draggable = true

    local UICorner = Instance.new("UICorner")
    UICorner.CornerRadius = UDim.new(0, 12)
    UICorner.Parent = OpenBtn

    local UIStroke = Instance.new("UIStroke")
    UIStroke.Parent = OpenBtn
    UIStroke.Color = Color3.fromRGB(0, 170, 255)
    UIStroke.Thickness = 1.5

    OpenBtn.MouseButton1Click:Connect(function()
        ToggleHub()
    end)
end

-- Ajuste de escala para Telas Pequenas (Mobile)
local function ApplyMobileScaling()
    task.spawn(function()
        task.wait(1)
        local OrionUI = CoreGui:FindFirstChild("Orion") or LocalPlayer.PlayerGui:FindFirstChild("Orion")
        if OrionUI then
            local MainFrame = OrionUI:FindFirstChild("Main", true)
            if MainFrame then
                local UIScale = MainFrame:FindFirstChildOfClass("UIScale") or Instance.new("UIScale")
                UIScale.Parent = MainFrame
                
                if UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled then
                    UIScale.Scale = 0.75
                else
                    UIScale.Scale = 1
                end
            end
        end
    end)
end

-- Inicialização
CreateMobileButton()
OrionLib:Init()
ApplyMobileScaling()
