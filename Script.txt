-- Verificação de ID dos Mares (Blox Fruits)
local ALLOWED_PLACES = {
    [2753915549] = true, -- First Sea
    [4442272183] = true, -- Second Sea
    [7449423635] = true  -- Third Sea
}

if not ALLOWED_PLACES[game.PlaceId] then
    pcall(function()
        game:GetService("StarterGui"):SetCore("SendNotification", {
            Title = "ZpeKeno Hub",
            Text = "Erro: Funciona apenas no Blox Fruits!",
            Duration = 5
        })
    end)
    return
end

-- Carregamento Seguro da Orion Library
local success, OrionLib = pcall(function()
    return loadstring(game:HttpGet("https://raw.githubusercontent.com/shlexware/Orion/main/source", true))()
end)

if not success or not OrionLib then
    warn("Falha ao carregar a Orion Library. Verifique a conexao do executor.")
    return
end

-- Criação da Janela Principal
local Window = OrionLib:MakeWindow({
    Name = "ZpeKeno Hub",
    HidePremium = false,
    SaveConfig = false,
    ConfigFolder = "ZpeKenoConfig",
    IntroText = "ZpeKeno Hub"
})

-- Variáveis de Configuração
local Settings = {
    Aimbot = false,
    AimbotDist = 500,
    ESP = false,
    SpeedHack = false,
    SpeedVal = 16
}

-- ABA PRINCIPAL
local MainTab = Window:MakeTab({
    Name = "Principal",
    Icon = "rbxassetid://4483345998",
    PremiumOnly = false
})

MainTab:AddSection({ Name = "Mira & Visuals" })

MainTab:AddToggle({
    Name = "Target Assist (Aimbot)",
    Default = false,
    Callback = function(Value)
        Settings.Aimbot = Value
    end    
})

MainTab:AddSlider({
    Name = "Distancia Maxima",
    Min = 100,
    Max = 1000,
    Default = 500,
    Color = Color3.fromRGB(0, 170, 255),
    Increment = 10,
    ValueName = "mstuds",
    Callback = function(Value)
        Settings.AimbotDist = Value
    end    
})

MainTab:AddToggle({
    Name = "Player Outline (ESP)",
    Default = false,
    Callback = function(Value)
        Settings.ESP = Value
    end    
})

-- ABA JOGADOR
local PlayerTab = Window:MakeTab({
    Name = "Jogador",
    Icon = "rbxassetid://4483345998",
    PremiumOnly = false
})

PlayerTab:AddSection({ Name = "Movimentacao" })

PlayerTab:AddToggle({
    Name = "Ativar SpeedHack",
    Default = false,
    Callback = function(Value)
        Settings.SpeedHack = Value
    end    
})

PlayerTab:AddSlider({
    Name = "Velocidade",
    Min = 16,
    Max = 100,
    Default = 16,
    Color = Color3.fromRGB(0, 255, 120),
    Increment = 1,
    ValueName = "Speed",
    Callback = function(Value)
        Settings.SpeedVal = Value
    end    
})

-- Inicializa a Interface
OrionLib:Init()
