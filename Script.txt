local function Create(className, props, children)
    local obj = Instance.new(className)
    for k, v in pairs(props or {}) do
        obj[k] = v
    end
    for _, child in ipairs(children or {}) do
        child.Parent = obj
    end
    return obj
end

local function AddCorner(parent, radius)
    return Create("UICorner", { CornerRadius = UDim.new(0, radius or 8), Parent = parent })
end

local function AddStroke(parent, color, thickness, transparency)
    return Create("UIStroke", {
        Color = color or Colors.Stroke,
        Thickness = thickness or 1,
        Transparency = transparency or 0.3,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = parent
    })
end

    { name = "OTHER",  icon = Icons.Other },
}

local tabBtns = {}

local tabCount = #tabNames
local tabWidth = 1 / tabCount

for i, tab in ipairs(tabNames) do
    local btn = Create("TextButton", {
        Name = tab.name .. "TabBtn",
        Size = UDim2.new(tabWidth, -6, 1, 0),
        Position = UDim2.new((i-1) * tabWidth, 3, 0, 0),
        BackgroundColor3 = (i == 1) and Colors.BgCard or Colors.BgPanel,
        BorderSizePixel = 0,
        Text = "",
        AutoButtonColor = false,
        Parent = TabBar
    })
    AddCorner(btn, 8)
    local stroke = AddStroke(btn, (i == 1) and Colors.Accent or Colors.Stroke, 1.5, (i == 1) and 0 or 0.5)

    local lbl = Create("TextLabel", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text = tab.icon .. "   " .. tab.name,
        TextColor3 = (i == 1) and Colors.Accent or Colors.TextSecondary,
        TextSize = 13,
        Font = Enum.Font.GothamBold,
        Parent = btn
    })

    tabBtns[tab.name] = { button = btn, stroke = stroke, label = lbl }

    local container = Create("Frame", {
        Name = tab.name .. "Content",
        Size = UDim2.new(1, -20, 1, -60),
        Position = UDim2.new(0, 10, 0, 60),
        BackgroundTransparency = 1,
        Visible = (i == 1),
        Parent = MainArea
    })
    tabContents[tab.name] = container
end

-- Troca de Aba
local function SwitchTab(tabName)
    for name, data in pairs(tabBtns) do
        if name == tabName then
            data.button.BackgroundColor3 = Colors.BgCard
            data.stroke.Color = Colors.Accent
            data.stroke.Transparency = 0
            data.label.TextColor3 = Colors.Accent
        else
            data.button.BackgroundColor3 = Colors.BgPanel
            data.stroke.Color = Colors.Stroke
            data.stroke.Transparency = 0.5
            data.label.TextColor3 = Colors.TextSecondary
        end
    end
    for name, container in pairs(tabContents) do
        container.Visible = (name == tabName)
    end
end

for name, data in pairs(tabBtns) do
    data.button.MouseButton1Click:Connect(function()
        SwitchTab(name)
    end)
end
