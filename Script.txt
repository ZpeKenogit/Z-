local function CreateSlider(parent, title, minVal, maxVal, defaultVal, callback)
    local card = Instance.new("Frame", parent)
    card.Size = UDim2.new(1, 0, 0, 50)
    card.BackgroundColor3 = C_CARD
    card.BorderSizePixel = 0
    Instance.new("UICorner", card).CornerRadius = UDim.new(0, 6)

    local lbl = Instance.new("TextLabel", card)
    lbl.Text = "<b>" .. title .. "</b>"
    lbl.RichText = true
    lbl.Font = Enum.Font.SourceSans
    lbl.TextSize = 14
    lbl.TextColor3 = C_TEXT
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Size = UDim2.new(0.5, 0, 0, 20)
    lbl.Position = UDim2.new(0, 10, 0, 5)
    lbl.BackgroundTransparency = 1

    local valBox = Instance.new("TextLabel", card)
    valBox.Text = tostring(defaultVal)
    valBox.Font = Enum.Font.SourceSansBold
    valBox.TextSize = 12
    valBox.TextColor3 = C_ACCENT
    valBox.Size = UDim2.new(0, 60, 0, 18)
    valBox.Position = UDim2.new(1, -70, 0, 5)
    valBox.BackgroundColor3 = Color3.fromRGB(15, 23, 40)
    Instance.new("UICorner", valBox).CornerRadius = UDim.new(0, 4)

    local sliderBg = Instance.new("TextButton", card)
    sliderBg.Text = ""
    sliderBg.Size = UDim2.new(1, -20, 0, 6)
    sliderBg.Position = UDim2.new(0, 10, 0, 32)
    sliderBg.BackgroundColor3 = Color3.fromRGB(40, 50, 70)
    Instance.new("UICorner", sliderBg).CornerRadius = UDim.new(1, 0)

    local fill = Instance.new("Frame", sliderBg)
    fill.Size = UDim2.new((defaultVal - minVal)/(maxVal - minVal), 0, 1, 0)
    fill.BackgroundColor3 = C_ACCENT
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

    local dragging = false
    local function UpdateSlider(input)
        local pos = math.clamp((input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X, 0, 1)
        local value = math.floor(minVal + (maxVal - minVal) * pos)
        fill.Size = UDim2.new(pos, 0, 1, 0)
        valBox.Text = tostring(value)
        callback(value)
    end

    sliderBg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            UpdateSlider(input)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            UpdateSlider(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

-- ==================== FUNCTIONAL FEATURES ENGINE ====================

-- Highlights & ESP Storage
local Highlights = {}

local function ClearVisuals()
    for _, h in pairs(Highlights) do
        if h then h:Destroy() end
    end
    Highlights = {}
end

-- ESP & Logic Loop
RunService.RenderStepped:Connect(function()
    -- Movement Loops
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
        local hum = LocalPlayer.Character.Humanoid
        if Settings.Other.MovementSpeed > 16 then hum.WalkSpeed = Settings.Other.MovementSpeed end
        if Settings.Other.JumpHeight > 50 then
            if hum.UseJumpPower then hum.JumpPower = Settings.Other.JumpHeight else hum.JumpHeight = Settings.Other.JumpHeight end
        end
    end

    -- Visual Highlights Engine
    if Settings.Visuals.PlayerOutline then
        for _, player in pairs(Players:GetPlayers()) do
            if player ~= LocalPlayer or not Settings.Visuals.ExcludeLocalPlayer then
                if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    if not Highlights[player] or not Highlights[player].Parent then
                        local hl = Instance.new("Highlight")
                        hl.Name = "ZpeKenoESP"
                        hl.FillColor = Color3.fromRGB(0, 255, 120)
