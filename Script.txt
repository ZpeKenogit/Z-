-- Trava de jogo (Blox Fruits: 2753915549)
local ALLOWED_PLACE_ID = 2753915549

if game.PlaceId ~= ALLOWED_PLACE_ID then
    pcall(function()
        game:GetService("StarterGui"):SetCore("SendNotification", {
            Title = "ZpeKeno Hub",
            Text = "Erro: Este script so funciona no Blox Fruits!",
            Duration = 6
        })
    end)
    return
end

-- Tenta carregar a biblioteca de interface
local OrionLib
local success, err = pcall(function()
    OrionLib = loadstring(game:HttpGet('https://raw.githubusercontent.com/jensonhirst/Orion/main/source'))()
end)

if not success or not OrionLib then
    warn("Falha ao carregar Orion Library: " .. tostring(err))
    return
end

-- Services
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local CoreGui = game:GetService("CoreGui")

local LocalPlayer = Players.LocalPlayer

-- Configurações padrão
local Settings = {
    Aimbot = { TargetAssist = false, TargetDistance = 500, Prediction = 50, TeamFilter = true },
    Visuals = { PlayerOutline = false, HealthPercent = false, Distance = false, ExcludeLocalPlayer = true },
    Other = { Dash = false, DashPower = 20, MovementSpeed = 16, JumpHeight = 50 }
}

-- Criar Janela Principal
local Window = OrionLib:MakeWindow({
    Name = "ZpeKeno | Game Development Hub",
    HidePremium = true,
    SaveConfig = false,
    IntroEnabled = true,
    IntroText = "ZpeKeno Hub"
})

-- ABA 1: AIMBOT
local AimbotTab = Window:MakeTab({ Name = "Aimbot", Icon = "rbxassetid://4483345998", PremiumOnly = false })
AimbotTab:AddSection({ Name = "AIMBOT - Target assistance and smart tracking" })

AimbotTab:AddToggle({
    Name = "Target Assist",
    Default = Settings.Aimbot.TargetAssist,
    Callback = function(Value) Settings.Aimbot.TargetAssist = Value end
})

AimbotTab:AddSlider({
    Name = "Target Distance",
    Min = 100, Max = 1000, Default = Settings.Aimbot.TargetDistance,
    Color = Color3.fromRGB(0, 170, 255), Increment = 10, ValueName = "m",
    Callback = function(Value) Settings.Aimbot.TargetDistance = Value end
})

AimbotTab:AddSlider({
    Name = "Prediction",
    Min = 1, Max = 100, Default = Settings.Aimbot.Prediction,
    Color = Color3.fromRGB(0, 170, 255), Increment = 1, ValueName = "%",
    Callback = function(Value) Settings.Aimbot.Prediction = Value end
})

AimbotTab:AddToggle({
    Name = "Team Filter",
    Default = Settings.Aimbot.TeamFilter,
    Callback = function(Value) Settings.Aimbot.TeamFilter = Value end
})

-- ABA 2: VISUAL
local VisualTab = Window:MakeTab({ Name = "Visual", Icon = "rbxassetid://4483345998", PremiumOnly = false })
VisualTab:AddSection({ Name = "VISUAL - Player information and highlighting" })

VisualTab:AddToggle({
    Name = "Player Outline",
    Default = Settings.Visuals.PlayerOutline,
    Callback = function(Value) Settings.Visuals.PlayerOutline = Value end
})

VisualTab:AddToggle({
    Name = "Health %",
    Default = Settings.Visuals.HealthPercent,
    Callback = function(Value) Settings.Visuals.HealthPercent = Value end
})

VisualTab:AddToggle({
    Name = "Distance",
    Default = Settings.Visuals.Distance,
    Callback = function(Value) Settings.Visuals.Distance = Value end
})

VisualTab:AddToggle({
    Name = "Exclude Local Player",
    Default = Settings.Visuals.ExcludeLocalPlayer,
    Callback = function(Value) Settings.Visuals.ExcludeLocalPlayer = Value end
})

-- ABA 3: OTHER
local OtherTab = Window:MakeTab({ Name = "Other", Icon = "rbxassetid://4483345998", PremiumOnly = false })
OtherTab:AddSection({ Name = "OTHER - Character and movement adjustments" })

OtherTab:AddSlider({
    Name = "Movement Speed",
    Min = 16, Max = 100, Default = Settings.Other.MovementSpeed,
    Color = Color3.fromRGB(0, 170, 255), Increment = 1, ValueName = "speed",
    Callback = function(Value) Settings.Other.MovementSpeed = Value end
})

OtherTab:AddSlider({
    Name = "Jump Height",
    Min = 50, Max = 200, Default = Settings.Other.JumpHeight,
    Color = Color3.fromRGB(0, 170, 255), Increment = 1, ValueName = "height",
    Callback = function(Value) Settings.Other.JumpHeight = Value end
})

-- Loop de atualização de mobilidade
RunService.RenderStepped:Connect(function()
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
        local Humanoid = LocalPlayer.Character.Humanoid
        if Settings.Other.MovementSpeed and Settings.Other.MovementSpeed > 16 then
            Humanoid.WalkSpeed = Settings.Other.MovementSpeed
        end
        if Settings.Other.JumpHeight and Settings.Other.JumpHeight > 50 then
            if Humanoid.UseJumpPower then
                Humanoid.JumpPower = Settings.Other.JumpHeight
            else
                Humanoid.JumpHeight = Settings.Other.JumpHeight
            end
        end
    end
end)

-- Função de alternar exibição
local function ToggleHub()
    local ui = CoreGui:FindFirstChild("Orion") or (LocalPlayer:FindFirstChild("PlayerGui") and LocalPlayer.PlayerGui:FindFirstChild("Orion"))
    if ui then
        ui.Enabled = not ui.Enabled
    end
end

-- Botão para Mobile
local function CreateMobileButton()
    local ScreenGui = Instance.new("ScreenGui")
    ScreenGui.Name = "ZpeKenoMobileUI"
    ScreenGui.ResetOnSpawn = false
    
    pcall(function() ScreenGui.Parent = CoreGui end)
    if not ScreenGui.Parent then
        ScreenGui.Parent = LocalPlayer:WaitForChild("PlayerGui")
    end

    local OpenBtn = Instance.new("TextButton")
    OpenBtn.Name = "ToggleZButton"
    OpenBtn.Parent = ScreenGui
    OpenBtn.Size = UDim2.new(0, 50, 0, 50)
    OpenBtn.Position = UDim2.new(0, 15, 0.35, 0)
    OpenBtn.BackgroundColor3 = Color3.fromRGB(15, 23, 42)
    OpenBtn.BorderColor3 = Color3.fromRGB(0, 170, 255)
    OpenBtn.BorderSizePixel = 2
    OpenBtn.Text = "Z"
    OpenBtn.TextColor3 = Color3.fromRGB(0, 170, 255)
    OpenBtn.TextSize = 22
    OpenBtn.Font = Enum.Font.SourceSansBold
    OpenBtn.Active = true
    OpenBtn.Draggable = true

    local UICorner = Instance.new("UICorner")
    UICorner.CornerRadius = UDim.new(0, 12)
    UICorner.Parent = OpenBtn

    OpenBtn.MouseButton1Click:Connect(function()
        ToggleHub()
    end)
end

-- Tecla 'Z' no teclado
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not gameProcessed and input.KeyCode == Enum.KeyCode.Z then
        ToggleHub()
    end
end)

-- Inicialização
CreateMobileButton()
OrionLib:Init()
